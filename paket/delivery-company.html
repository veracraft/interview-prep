package main

import (
	"errors"
	"fmt"
	"sort"
	"sync"
	"time"
)

// Assumptions (agreed with interviewer):
// - in-memory, single process; methods are safe for concurrent use
// - quantities are ints > 0; money is int64 cents

// ---------- domain types ----------
type (
	SKU     string
	StoreID string
	OrderID string
)

type Status string

const (
	StatusOpen      Status = "OPEN"
	StatusDone      Status = "DONE"
	StatusCancelled Status = "CANCELLED"
)

// one place for valid transitions
var transitions = map[Status][]Status{
	StatusOpen: {StatusDone, StatusCancelled},
}

func canMove(from, to Status) bool {
	for _, s := range transitions[from] {
		if s == to {
			return true
		}
	}
	return false
}

// ---------- errors ----------
var (
	ErrInvalidArgument = errors.New("invalid argument")
	ErrNotFound        = errors.New("not found")
	ErrConflict        = errors.New("conflict")
	ErrInvalidState    = errors.New("invalid state")
)

type ShortageError struct {
	SKU                  SKU
	Requested, Available int
}

func (e *ShortageError) Error() string {
	return fmt.Sprintf("shortage %s: requested %d, available %d", e.SKU, e.Requested, e.Available)
}

// ---------- time is injected ----------
type Clock interface{ Now() time.Time }
type realClock struct{}

func (realClock) Now() time.Time { return time.Now() }

type fakeClock struct{ t time.Time }

func (f *fakeClock) Now() time.Time          { return f.t }
func (f *fakeClock) Advance(d time.Duration) { f.t = f.t.Add(d) }
// ---------- service ----------
type Service struct {
	mu    sync.Mutex
	clock Clock
	stock map[StoreID]map[SKU]int
	// orders map[OrderID]*order ...
}

func NewService(c Clock) *Service {
	if c == nil {
		c = realClock{}
	}
	return &Service{clock: c, stock: map[StoreID]map[SKU]int{}}
}

func (s *Service) Receive(store StoreID, sku SKU, qty int) error {
	if store == "" || sku == "" || qty <= 0 {
		return fmt.Errorf("%w: store=%q sku=%q qty=%d",
			ErrInvalidArgument, store, sku, qty)
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.stock[store] == nil {
		s.stock[store] = map[SKU]int{}
	}
	s.stock[store][sku] += qty
	return nil
}

// deterministic output: sort keys
func (s *Service) Snapshot(store StoreID) []string {
	s.mu.Lock()
	defer s.mu.Unlock()
	skus := make([]SKU, 0, len(s.stock[store]))
	for k := range s.stock[store] {
		skus = append(skus, k)
	}
	sort.Slice(skus, func(i, j int) bool { return skus[i] < skus[j] })
	out := make([]string, 0, len(skus))
	for _, k := range skus {
		out = append(out, fmt.Sprintf("%s=%d", k, s.stock[store][k]))
	}
	return out
}

// ---------- demo + mini tests ----------
func check(name string, ok bool) {
	if ok {
		fmt.Println("PASS", name)
	} else {
		fmt.Println("FAIL", name)
	}
}

func main() {
	clk := &fakeClock{t: time.Date(2026, 11, 3,
		12, 0, 0, 0, time.UTC)}
	s := NewService(clk)

	err := s.Receive("hel-1", "milk", 0)
	check("zero qty rejected",
		errors.Is(err, ErrInvalidArgument))

	check("receive ok",
		s.Receive("hel-1", "milk", 5) == nil)
	fmt.Println(s.Snapshot("hel-1"))

	// typed error example:
	var se *ShortageError
	err = fmt.Errorf("reserve: %w",
		&ShortageError{"milk", 9, 5})
	check("errors.As works", errors.As(err, &se) &&
		se.Available == 5)
}

Part 5: "How would this work in production?"

Persistence: "State goes to Postgres. Reservations become rows; the check-and-apply runs in one transaction with SELECT … FOR UPDATE on the stock rows, or a conditional update: UPDATE stock SET reserved = reserved + $q WHERE … AND on_hand - reserved >= $q. Idempotency becomes a unique constraint on order_id."

Eşzamanlılık / ölçek: "One mutex serializes everything. Next step is a lock per store, because stores are independent. At scale, partition by store: one Kafka partition or one worker owns a store, so there's a single writer and no cross-node locks."

Event'ler: "Every state change emits an event, like ReservationCreated or StockReceived, through a pending_events table written in the same transaction (the transactional outbox pattern), so the DB write and the event can't diverge. Consumers are idempotent because delivery is at-least-once."

Sırasız / tekrarlı event: event_id ile dedupe, entity başına sequence/version; eski version'u yok say; boşluk varsa kısa buffer + timeout, sonra reconcile.

API: POST /stores/{id}/reservations + Idempotency-Key → 201; stok yok → 409 + eksik listesi; geçersiz → 400/422. gRPC'de FAILED_PRECONDITION, ALREADY_EXISTS.

Gözlemlenebilirlik: reservation başarı/red oranı, shortage oranı SKU bazlı, p99 latency, expired reservation sayısı, pending_events lag (en eski gönderilmemiş satırın yaşı). Log'da order_id, store_id.

Test: table-driven unit testler, fake clock, go test -race, eşzamanlı rezervasyon testi (100 goroutine, son ürün tek kişiye), DB ile entegrasyon testi (testcontainers).

Expiry production'da: lazy kontrol + periyodik sweeper job (ya da DB'de expires_at index + batch update). Saat kayması: tek kaynak (DB'nin now()).

Para / fiyat: fiyatı sipariş anında snapshot'la, sonradan değişen fiyat geçmiş siparişi değiştirmesin.

AI kullanımı (ilan istiyor; kendi pratiğine göre uyarla): "I use AI assistants daily, for boilerplate, tests and exploring unfamiliar code, but I review and run everything and I own the result. Domain rules like rounding or state transitions I write myself, and I use AI to challenge the edge cases."

Isınma: tek mağaza stok rezervasyonu (kağıtta 15 dk)6 / 14 · Isınma
Problem: Tek mağaza, in-memory servis. Bir sipariş için stok rezerve et ve rezervasyonu iptal et. İstekler tekrar gelebilir. Hata durumunda state değişmemeli. (Alıştırma kuralları; şirketten bildirilmiş bir soru değil.)

Sözleşme

  • Bir order ID = bir SKU + pozitif tam sayı miktar.
  • Aynı ID + aynı girdi → mevcut rezervasyonu dön. Aynı ID + farklı girdi → conflict.
  • İptal stoğu bir kez geri verir; tekrar iptal etkisizdir.
  • İptal edilmiş bir rezervasyonun tekrarı, iptal kaydını döner; yeniden rezerve etmez.
  • Bilinmeyen iptal hata; olmayan SKU = 0 stok. Tek process, sıralı çağrılar, süre dolması yok.

Go cevabı (derlendi, go test -race geçti)

package main

import "errors"

type Reservation struct {
  SKU   string
  Qty   int
  State string
}
type Store struct {
  stock  map[string]int
  orders map[string]Reservation
}

func NewStore(stock map[string]int) *Store {
  copyStock := make(map[string]int, len(stock))
  for sku, qty := range stock {
    copyStock[sku] = qty
  }
  return &Store{copyStock, map[string]Reservation{}}
}
func (s *Store) Reserve(id, sku string, qty int) (Reservation, error) {
  if id == "" || sku == "" || qty <= 0 {
    return Reservation{}, errors.New("invalid input")
  }
  if old, ok := s.orders[id]; ok {
    if old.SKU != sku || old.Qty != qty {
      return Reservation{}, errors.New("conflicting replay")
    }
    return old, nil
  }
  if s.stock[sku] < qty {
    return Reservation{}, errors.New("insufficient stock")
  }
  row := Reservation{sku, qty, "RESERVED"}
  s.stock[sku] -= qty
  s.orders[id] = row
  return row, nil
}
func (s *Store) Cancel(id string) (Reservation, error) {
  old, ok := s.orders[id]
  if !ok {
    return Reservation{}, errors.New("unknown order")
  }
  if old.State == "CANCELLED" {
    return old, nil
  }
  s.stock[old.SKU] += old.Qty
  old.State = "CANCELLED"
  s.orders[id] = old
  return old, nil
}

Dönüşler değer (struct kopyası), çağıran iç kaydı değiştiremez. Bu çekirdek sıralı çağrı varsayar; eşzamanlı çağrıda tüm Reserve/Cancel işlemini kilitle. Birden çok replika → DB transaction veya başka paylaşılan atomik otorite.

The main guarantee is that a repeated request can't change stock twice. I check the existing order before availability, and I keep each state transition together.

    return BootstrapResponse{}, ctx.Err()
  }
  if err != nil {
    return BootstrapResponse{}, err
  }
  var wg sync.WaitGroup
  var payment Payment
  var address Address
  var paymentErr, addressErr error
  wg.Add(2)

  go func() {
    defer wg.Done()
    paymentCtx, cancel := context.WithTimeout(
      ctx, 500*time.Millisecond,
    )
    defer cancel()
    payment, paymentErr = s.Payment.GetPayment(
      paymentCtx, consumer.ID,
    )
  }()
  go func() {
    defer wg.Done()
    address, addressErr = s.Address.GetAddress(
      ctx, consumer.ID,
    )
  }()
  wg.Wait()

  if err := ctx.Err(); err != nil {
    return BootstrapResponse{}, err
  }
  // One goroutine builds the response from these local results.
  // Treat an optional error as partial data, not total failure.
  response := BootstrapResponse{Consumer: &consumer}
  if paymentErr == nil {
    response.Payment = &payment
  }
  if addressErr == nil {
    response.Address = &address
  }
  response.Partial = paymentErr != nil || addressErr != nil
  return response, nil
}

Sınırı tam olarak anlat

Payment gets its own child context with a 500 ms timeout. Address keeps the parent context. A Payment timeout becomes a partial response, not a failure. This relies on the Payment client respecting cancellation: a context doesn't kill a goroutine.

  • WaitGroup tamamlanmayı bekler; context'e uymayan bir client yine bloklayabilir. Transport ve client deadline'a uymalı.
  • Parent iptali ile opsiyonel Payment timeout'u farklı şeyler: parent iptal → hata dön; Payment timeout → kısmi cevap.
  • Paylaşılan state yok: her goroutine kendi değişkenine yazar, cevap wg.Wait()'ten sonra kurulur → race yok.

Kısa retry cevabı

I retry transient failures only when repeating the call is safe. I cap total attempts, back off with jitter, and stay inside the dependency's time budget. If most calls are failing, I'd rather shed load with concurrency limits and a circuit breaker. And I keep the retry policy in one layer, so retries don't multiply.

Gösterilecek testler

  • Consumer hatası → opsiyonel çağrılar yapılmaz.
  • Bir opsiyonel hata, diğer sonucu korur (Partial = true).
  • Parent iptali yayılır.
  • Payment timeout'u Address'i iptal etmez.
  • ctx.Done()'u bekleyen fake client ile timeout testi; go test -race.
Hatırlatma: önce doğru ve race-safe çözüm; context, timeout ve retry sonra, sorulunca. Erken soyutlama yok.

Drill 1 · Pick

func (s *Store) Pick(o OrderID, sku SKU, qty int) ([]Allocation, error) {
  if o == "" || qty <= 0 {
    return nil, fmt.Errorf("%w: order=%q qty=%d", ErrInvalidArgument, o, qty)
  }
  s.mu.Lock(); defer s.mu.Unlock()
  if p, ok := s.picks[o]; ok {                     // idempotent replay
    if p.sku != sku || p.qty != qty { return nil, ErrConflict }
    return append([]Allocation(nil), p.allocs...), nil
  }
  now := s.clock.Now()
  avail := 0                                         // 1) check
  for _, l := range s.lots[sku] {
    if now.Before(l.Expiry) { avail += l.Qty }
  }
  if avail < qty {
    return nil, &ShortageError{SKU: sku, Requested: qty, Available: avail}
  }
  var out []Allocation                               // 2) apply (FEFO)
  left := qty
  for _, l := range s.lots[sku] {
    if left == 0 { break }
    if !now.Before(l.Expiry) || l.Qty == 0 { continue }
    take := min(l.Qty, left)
    l.Qty -= take; left -= take
    out = append(out, Allocation{LotID: l.ID, Qty: take})
  }
  s.picks[o] = pickRecord{sku, qty, out}
  return append([]Allocation(nil), out...), nil
}

Drill 2 · NextTask (lazy timeout + O(n) seçim)

func (s *Service) NextTask(p PickerID) (Task, error) {
  if p == "" { return Task{}, ErrInvalidArgument }
  s.mu.Lock(); defer s.mu.Unlock()
  now := s.clock.Now()
  s.expireLocked(now)                    // ASSIGNED && now >= assignedAt+15m → QUEUED

  if id, ok := s.byPicker[p]; ok {       // already has one: return it
    return *s.tasks[id], nil
  }
  var best *Task
  for _, t := range s.tasks {
    if t.Status != StatusQueued { continue }
    if best == nil || s.policy.Less(t, best) { best = t }
  }
  if best == nil { return Task{}, ErrNoTask }
  best.Status, best.Picker, best.AssignedAt = StatusAssigned, p, now
  s.byPicker[p] = best.ID
  return *best, nil                      // copy, not pointer
}

type Policy interface{ Less(a, b *Task) bool }
type EarliestPromise struct{}
func (EarliestPromise) Less(a, b *Task) bool {
  if !a.PromisedAt.Equal(b.PromisedAt) { return a.PromisedAt.Before(b.PromisedAt) }
  return a.Seq < b.Seq
}
type ExpressFirst struct{ Next Policy }   // decorator / chain
func (e ExpressFirst) Less(a, b *Task) bool {
  if a.Express != b.Express { return a.Express }
  return e.Next.Less(a, b)
}

func (s *Service) Assign(wid WorkerID, sid ShiftID) error {
  s.mu.Lock(); defer s.mu.Unlock()
  w, ok := s.workers[wid]
  if !ok { return fmt.Errorf("%w: worker %s", ErrNotFound, wid) }
  sh, ok := s.shifts[sid]
  if !ok { return fmt.Errorf("%w: shift %s", ErrNotFound, sid) }
  if s.assigned[sid][wid] { return nil }            // idempotent
  if len(s.assigned[sid]) >= sh.Capacity {
    return fmt.Errorf("%w: shift %s is full", ErrInvalidState, sid)
  }
  ctx := RuleContext{w, sh, s.shiftsOfLocked(wid)}
  var vs []Violation
  for _, r := range rulesByCountry[s.countryOf(sh.Store)] {
    if v := r.Check(ctx); v != nil { vs = append(vs, *v) }
  }
  if len(vs) > 0 {
    sort.Slice(vs, func(i, j int) bool { return vs[i].Rule < vs[j].Rule })
    return &ValidationError{Violations: vs}
  }
  if s.assigned[sid] == nil { s.assigned[sid] = map[WorkerID]bool{} }
  s.assigned[sid][wid] = true
  return nil
}

MinRest ve MaxWeekly

type MinRest struct{ Min time.Duration }
func (MinRest) Name() string { return "min_rest" }
func (r MinRest) Check(c RuleContext) *Violation {
  for _, e := range c.Existing {
    var gap time.Duration
    switch {
    case !e.End.After(c.Shift.Start): gap = c.Shift.Start.Sub(e.End) // e before
    case !c.Shift.End.After(e.Start): gap = e.Start.Sub(c.Shift.End) // e after
    default: continue                         // overlap: NoOverlap reports it
    }
    if gap < r.Min {
      return &Violation{"min_rest", fmt.Sprintf("only %v rest around shift %s", gap, e.ID)}
    }
  }
  return nil
}

type MaxWeekly struct{ Max time.Duration }
func (MaxWeekly) Name() string { return "max_weekly_hours" }
func (r MaxWeekly) Check(c RuleContext) *Violation {
  y, w := c.Shift.Start.ISOWeek()
  total := c.Shift.End.Sub(c.Shift.Start)
  for _, e := range c.Existing {
    if ey, ew := e.Start.ISOWeek(); ey == y && ew == w {
      total += e.End.Sub(e.Start)
    }
  }
  if total > r.Max {
    return &Violation{"max_weekly_hours", fmt.Sprintf("%v > %v", total, r.Max)}
  }
  return nil
}
